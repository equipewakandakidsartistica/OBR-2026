def iniciar_robo():
    global processo_robo
 
    # Verificar se já está rodando
 
print()
    print("======================================")
    print("🔴 BOTÃO DESLIGAR PRESSIONADO")
    print("======================================")
 
    # Parar programa de visão
    if processo_robo is not None and processo_robo.poll() is None:
 
        print("Parando sistema de visão...")
 
        processo_robo.terminate()
 
        try:
            processo_robo.wait(timeout=5)
        except subprocess.TimeoutExpired:
            processo_robo.kill()
 
        print("Sistema de visão parado.")
 
    else:
        print("Sistema de visão já estava parado.")
 
    print("Desligando Raspberry Pi...")
    time.sleep(1)
 
    subprocess.Popen([
        "sudo",
        "shutdown",
        "-h",
        "now"
])
